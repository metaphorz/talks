Classify Segments by NLCD + Depth → Fetch or Obstruction

// Classify each segment as fetch (wave growth) or obstruction (wave attenuation)
// Classification depends on NLCD code AND water depth
function classifyFetchOrObstruction(lulc, depth, stillWater, category) {
  // Water (11, 12) → always FETCH
  if (lulc === 11 || lulc === 12) return 'fetch';

  // Buildings (22-24), Forest (41-43), Dunes (31) → always obstruction
  if ((lulc >= 22 && lulc <= 24) || (lulc >= 41 && lulc <= 43) || lulc === 31) {
    return 'obstruction';
  }

  // Scrub (51-52): obstruction if depth < 5 ft, fetch if depth ≥ 5 ft
  if (lulc === 51 || lulc === 52) {
    return depth >= 5 ? 'fetch' : 'obstruction';
  }

  // Grassland (71-74, 81-82): fetch if depth ≥ 3 ft, obstruction otherwise
  if ((lulc >= 71 && lulc <= 74) || lulc === 81 || lulc === 82) {
    return depth >= 3 ? 'fetch' : 'obstruction';
  }

  // Wetlands (90, 95): fetch if still water ≥ 4 ft MSL, obstruction otherwise
  if (lulc === 90 || lulc === 95) {
    return stillWater >= 4 ? 'fetch' : 'obstruction';
  }

  return 'unknown';
}

Apply NAS Eq. 13 Wave Growth: Hf = √[(G·d)² + Hi²]

// For FETCH segments: waves grow using NAS Equation 13
if (segment.fetchType === 'fetch') {
  // NAS Equation 13: Inland fetch wave growth
  // Hf = √[(G×d)² + Hi²]
  // G = growth factor interpolated from fetch length (0.0-5.0 mi → 0.0-0.5)
  // d = average depth in segment (feet)
  // Hi = wave height at segment start
  // Hf = wave height at segment end

  var fetchLengthMiles = segment.lengthFeet / 5280;
  var G = interpolateGCurve(fetchLengthMiles);
  var avgDepth = segment.avgDepth;
  var Hi = waveHeightAtStart;

  var growthTerm = G * avgDepth;
  var Hf = Math.sqrt(growthTerm * growthTerm + Hi * Hi);

  hxAfter = Hf;  // Wave can grow larger in fetch areas
  console.log('→ FETCH: NAS Eq 13, G=' + G.toFixed(3) + ', Hf=' + Hf.toFixed(2));
}

Apply NAS Eq. 5-7 Dune Transmission

// For DUNE_BARRIER segments: natural barrier transmission (no overtopping)
if (segment.category === 'dune_barrier') {
  // NAS Equations 5-7: Natural barrier (dunes, bars, breakwaters)
  var Z_b = barrierElevationSum / barrierElevationCount;  // Average barrier elevation
  var S_b = waveInputs.stillWater;                         // Still water level
  var d_b = S_b - Z_b;                                     // Depth at barrier
  var H_i = waveHeightAtStart;                             // Incident wave height

  if (Z_b > S_b) {
    // Equation 7: Barrier above still water → no transmission
    hxAfter = 0;
  } else if (d_b > 0) {
    var breakingHeightAtBarrier = 0.78 * d_b;
    if (H_i < breakingHeightAtBarrier) {
      // Equation 5: Full transmission (wave not breaking)
      hxAfter = H_i;
    } else {
      // Equation 6: Limited transmission (wave breaking on barrier)
      hxAfter = breakingHeightAtBarrier;  // H_t = 0.78 × d_b
    }
  }
  console.log('→ DUNE: NAS Eq 5-7, Z_b=' + Z_b.toFixed(1) + ', Hf=' + hxAfter.toFixed(2));
}

Apply NAS Eq. 8-10 Seawall Transmission

// For ELONGATED_BARRIER segments: seawalls/levees (allows partial overtopping)
if (segment.category === 'elongated_barrier') {
  // NAS Equations 8-10: Elongated barriers (seawalls, levees, revetments)
  var Z_b = barrierElevationSum / barrierElevationCount;
  var S_b = waveInputs.stillWater;
  var d_b = S_b - Z_b;
  var H_i = waveHeightAtStart;

  if (Z_b > (S_b + 0.5 * H_i)) {
    // Equation 10: Barrier above still water + half wave height → no transmission
    hxAfter = 0;
  } else if (d_b > 0) {
    var breakingHeightAtBarrier = 0.78 * d_b;
    if (H_i < breakingHeightAtBarrier) {
      // Equation 8: Full transmission
      hxAfter = H_i;
    } else {
      // Equation 9: Partial transmission with overtopping
      // H_t = 0.78×d_b + 0.5×H_i
      hxAfter = breakingHeightAtBarrier + 0.5 * H_i;
    }
  }
  console.log('→ SEAWALL: NAS Eq 8-10, Hf=' + hxAfter.toFixed(2));
}

Apply NAS Eq. 11 Vegetation Drag

// For VEGETATION obstruction segments: drag force transmission
if (segment.category && VEGETATION_DAMPING_LOOKUP[segment.category]) {
  // NAS Equation 11: Vegetation transmission
  // B = 1 / (1 + (C_D·H_i·h·D·w) / (3π·b²·d²))
  // C_D = drag coefficient
  // h = wetted height = min(actual_height, depth)
  // D = effective stem diameter (ft)
  // b = horizontal spacing between stems (ft)
  // w = segment width along transect (ft)
  // d = average water depth (ft)

  var veg = VEGETATION_DAMPING_LOOKUP[segment.category];
  var widthFeet = Math.max(veg.minWidth || 0, segment.lengthFeet);
  var wettedHeight = Math.min(veg.h, avgDepth);  // Proper submergence
  var denominator = 3 * Math.PI * veg.b * veg.b * avgDepth * avgDepth;

  if (denominator > 0 && waveHeightAtStart > 0) {
    var reduction = 1 + (veg.C_D * waveHeightAtStart * wettedHeight * veg.D * widthFeet) / denominator;
    if (reduction > 0) {
      hxAfter = waveHeightAtStart / reduction;  // H_t = B × H_i
      var B = 1 / reduction;
      console.log('→ VEGETATION: NAS Eq 11, B=' + B.toFixed(3) + ', Hf=' + hxAfter.toFixed(2));
    }
  }
}

Apply NAS Eq. 12 Building Transmission

// For BUILDING obstruction segments: exponential transmission per row
if (segment.category && BUILDING_DAMPING_LOOKUP[segment.category]) {
  // NAS Equation 12: Building transmission
  // B = r^(n/2)
  // r = open space ratio between buildings (0.25 high, 0.4 medium, 0.6 low density)
  // n = number of building rows = segment_width / row_spacing

  var building = BUILDING_DAMPING_LOOKUP[segment.category];
  var rows = Math.max(1, Math.round(segment.lengthFeet / building.rowSpacing));  // n
  var transmission = Math.pow(building.r, rows / 2);  // B = r^(n/2)
  hxAfter = waveHeightAtStart * transmission;          // H_t = B × H_i

  console.log('→ BUILDINGS: NAS Eq 12, n=' + rows + ' rows, r=' + building.r + ', B=' + transmission.toFixed(3) + ', Hf=' + hxAfter.toFixed(2));
}
